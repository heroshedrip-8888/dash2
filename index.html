    <script>

        const SHEET_PUB_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vRE8mWvyuqQFvXxjCwE8h1zeFVv1yiJes0cp7I4Y6mbnLkWdOYGj66EKxwfgpDLD3HpLNu3uar9ayyq/pubhtml";


        async function loadSheetData() {

            try {

                const response = await fetch(SHEET_PUB_URL);

                const text = await response.text();

                const parser = new DOMParser();

                const doc = parser.parseFromString(text, "text/html");

                const tables = doc.querySelectorAll('table');


                const characters = [];

                tables.forEach((table, index) => {

                    const rows = table.querySelectorAll('tr');

                    if (rows.length < 3) return;


                    const name = rows[0].cells[0]?.textContent.trim() || `Character ${index+1}`;

                    const data = { name, views: [], followers: [], engagement: [] };


                    for (let i = 2; i < Math.min(14, rows.length); i++) {

                        const cells = rows[i].cells;

                        if (cells.length > 2) {

                            data.views.push(parseInt(cells[1]?.textContent) || 0);

                            data.followers.push(parseInt(cells[2]?.textContent) || 0);

                            data.engagement.push(parseInt(cells[3]?.textContent) || 0);

                        }

                    }

                    characters.push(data);

                });


                renderDashboard(characters);

            } catch(e) {

                console.error("Data load error", e);

                alert("Could not load live data. Using demo data.");

                renderDashboard([

                    {name:"Miranda Kiera", views:Array(12).fill(800), followers:Array(12).fill(200), engagement:Array(12).fill(60834)},

                    {name:"Ava Sinclair", views:Array(12).fill(800), followers:Array(12).fill(200), engagement:Array(12).fill(60834)},

                    {name:"Kane Voss", views:Array(12).fill(800), followers:Array(12).fill(200), engagement:Array(12).fill(60834)}

                ]);

            }

        }


        function renderDashboard(characters) {

            // Global Stats

            let totalV = 0, totalF = 0, totalE = 0;

            characters.forEach(c => {

                totalV += c.views.reduce((a,b)=>a+b,0);

                totalF += c.followers.reduce((a,b)=>a+b,0);

                totalE += c.engagement.reduce((a,b)=>a+b,0);

            });


            document.getElementById('global-stats').innerHTML = `

                <div class="glass rounded-3xl p-8">

                    <p class="text-slate-400">TOTAL VIEWS</p>

                    <p class="text-6xl font-bold text-white">${totalV.toLocaleString()}</p>

                </div>

                <div class="glass rounded-3xl p-8">

                    <p class="text-slate-400">TOTAL FOLLOWERS</p>

                    <p class="text-6xl font-bold text-white">${totalF.toLocaleString()}</p>

                </div>

                <div class="glass rounded-3xl p-8">

                    <p class="text-slate-400">TOTAL ENGAGEMENT</p>

                    <p class="text-6xl font-bold text-white">${totalE.toLocaleString()}</p>

                </div>

                <div class="glass rounded-3xl p-8">

                    <p class="text-slate-400">CHARACTERS</p>

                    <p class="text-6xl font-bold text-white">${characters.length}</p>

                </div>

            `;


            // Characters

            const container = document.getElementById('characters-container');

            container.innerHTML = '';

            characters.forEach((char, i) => {

                const card = document.createElement('div');

                card.className = "glass rounded-3xl p-8 card-hover";

                card.innerHTML = `

                    <div class="flex gap-4 items-center mb-6">

                        <div class="w-12 h-12 bg-gradient-to-br from-cyan-400 to-blue-600 rounded-2xl flex items-center justify-center text-white text-2xl font-bold">${char.name[0]}</div>

                        <div class="font-semibold text-2xl">${char.name}</div>

                    </div>

                    <div class="grid grid-cols-3 gap-6">

                        <div><div class="text-xs text-slate-400">VIEWS</div><div class="text-3xl font-bold">${char.views.reduce((a,b)=>a+b,0)}</div></div>

                        <div><div class="text-xs text-slate-400">FOLLOWERS</div><div class="text-3xl font-bold">${char.followers.reduce((a,b)=>a+b,0)}</div></div>

                        <div><div class="text-xs text-slate-400">ENGAGE</div><div class="text-3xl font-bold">${(char.engagement.reduce((a,b)=>a+b,0)/1000).toFixed(0)}K</div></div>

                    </div>

                `;

                container.appendChild(card);

            });


            document.getElementById('last-updated').innerHTML = `✅ Updated just now`;

        }


        // Load data

        window.onload = loadSheetData;
